// ===================================================== // STATIC FILES // ===================================================== // index.html, music folder, css/js etc. থাকবে public folder-এর ভিতরে app.use(express.static(path.join(__dirname, "public"))); // ===================================================== // GAME STATE // ===================================================== const state = {}; for (const country of Object.keys(info)) { state[country] = { score: 0, commentCount: 0, superChats: 0, latestCommenter: "" }; } // ===================================================== // PLAYER STATE // ===================================================== const players = {}; // ===================================================== // EVENT STATE // ===================================================== let eventCounter = 0; let lastEvent = { id: "", country: "", type: "", displayName: "", points: 0, message: "", amountMicros: 0, currency: "", time: 0 }; // ===================================================== // FIND COUNTRY // ===================================================== function findCountryInText(message) { const text = normalize(message); if (!text) { return null; } for (const item of countrySearchList) { const escaped = item.alias.replace( /[.*+?^${}()|[\]\\]/g, "\\$&" ); const regex = new RegExp( "(^|\\s)" + escaped + "(?=\\s|$)", "i" ); if (regex.test(text)) { return item.country; } } return null; } // ===================================================== // YOUTUBE LIVE CHAT // ===================================================== let liveChatId = null; let nextPageToken = null; let pollingTimer = null; const seenMessageIds = new Set(); // ===================================================== // GET LIVE CHAT ID // ===================================================== async function getLiveChatId() { try { const response = await youtube.videos.list({ part: ["liveStreamingDetails"], id: [VIDEO_ID] }); const item = response.data.items && response.data.items[0]; if (!item) { throw new Error( "YouTube video not found." ); } const details = item.liveStreamingDetails; if (!details) { throw new Error( "This video is not a live stream." ); } liveChatId = details.activeLiveChatId || null; if (!liveChatId) { throw new Error( "Live chat is not active." ); } nextPageToken = null; console.log( "LIVE CHAT CONNECTED:", liveChatId ); } catch (error) { liveChatId = null; console.error( "GET LIVE CHAT ERROR:", error.response?.data || error.message ); throw error; } } // ===================================================== // CREATE / GET PLAYER // ===================================================== function getPlayer(displayName) { const safeName = String( displayName || "Anonymous" ).trim(); const key = safeName.toLowerCase(); if (!players[key]) { players[key] = { key, displayName: safeName || "Anonymous", country: "", score: 0, comments: 0, superChats: 0 }; } return players[key]; } // ===================================================== // PROCESS YOUTUBE MESSAGE // ===================================================== function processMessage(item) { try { if (!item || !item.id) { return; } // Prevent duplicate messages if ( seenMessageIds.has(item.id) ) { return; } seenMessageIds.add(item.id); // Keep memory under control if ( seenMessageIds.size > 5000 ) { const firstId = seenMessageIds .values() .next() .value; seenMessageIds.delete( firstId ); } const snippet = item.snippet || {}; const author = item.authorDetails || {}; const displayName = author.displayName || "Anonymous"; const type = snippet.type || ""; let message = ""; let amountMicros = 0; let currency = ""; // ================================================= // NORMAL COMMENT // ================================================= if ( type === "textMessageEvent" ) { message = snippet.displayMessage || snippet.textMessageDetails ?.messageText || ""; } // ================================================= // SUPER CHAT // ================================================= if ( type === "superChatEvent" ) { const details = snippet.superChatDetails || {}; message = details.userComment || ""; amountMicros = Number( details.amountMicros || 0 ); currency = details.currency || "USD"; } // Ignore unsupported events if ( !message && type !== "superChatEvent" ) { return; } console.log( "YOUTUBE:", displayName, "=>", message ); // ================================================= // FIND COUNTRY // ================================================= const country = findCountryInText( message ); if (!country) { console.log( "NO COUNTRY FOUND:", message ); return; } // ================================================= // NORMAL COMMENT // ================================================= if ( type === "textMessageEvent" ) { const points = COMMENT_POINTS; state[country].score += points; state[country].commentCount += 1; state[country].latestCommenter = displayName; const player = getPlayer( displayName ); player.country = country; player.score += points; player.comments += 1; eventCounter++; lastEvent = { id: `${item.id}-${eventCounter}`, country, type: "comment", displayName, points, message, amountMicros: 0, currency: "", time: Date.now() }; console.log( `COMMENT +${points} | ${displayName} | ${country}` ); return; } // ================================================= // SUPER CHAT // ================================================= if ( type === "superChatEvent" ) { const amountUSD = amountMicros / 1000000; const points = Math.max( 1, Math.round( amountUSD * SUPERCHAT_POINTS_PER_USD ) ); state[country].score += points; state[country].superChats += 1; state[country].latestCommenter = displayName; const player = getPlayer( displayName ); player.country = country; player.score += points; player.superChats += 1; eventCounter++; lastEvent = { id: `${item.id}-${eventCounter}`, country, type: "superchat", displayName, points, message, amountMicros, currency, time: Date.now() }; console.log( `SUPER CHAT +${points} | ${displayName} | ${country} | ${amountUSD} ${currency}` ); } } catch (error) { console.error( "PROCESS MESSAGE ERROR:", error.message ); } } // ===================================================== // POLL YOUTUBE CHAT // ===================================================== async function pollChat() { try { if (!liveChatId) { await getLiveChatId(); } const params = { liveChatId, part: [ "snippet", "authorDetails" ], maxResults: 200 }; if (nextPageToken) { params.pageToken = nextPageToken; } const response = await youtube .liveChatMessages .list(params); nextPageToken = response.data.nextPageToken || null; const messages = response.data.items || []; console.log( "YouTube messages:", messages.length ); for ( const item of messages ) { processMessage(item); } const wait = response.data .pollingIntervalMillis || 5000; schedulePoll(wait); } catch (error) { console.error( "YOUTUBE CHAT ERROR:", error.response?.data || error.message ); liveChatId = null; nextPageToken = null; schedulePoll(10000); } } // ===================================================== // SCHEDULE NEXT POLL // ===================================================== function schedulePoll( milliseconds ) { if (pollingTimer) { clearTimeout( pollingTimer ); } pollingTimer = setTimeout( pollChat, Math.max( milliseconds || 5000, 1000 ) ); } // ===================================================== // API STATE // ===================================================== app.get( "/api/state", (req, res) => { const result = {}; // ================================================= // ALL COUNTRIES // ================================================= for ( const country of Object.keys(info) ) { const item = state[country] || { score: 0, commentCount: 0, superChats: 0, latestCommenter: "" }; result[country] = { key: country, name: info[country][1], flag: info[country][0], score: item.score || 0, commentCount: item.commentCount || 0, superChats: item.superChats || 0, latestCommenter: item.latestCommenter || "" }; } // ================================================= // COUNTRY LEADERBOARD // ================================================= const activeCountries = Object.entries(result) .filter( ([, item]) => item.score > 0 ) .sort( (a, b) => b[1].score - a[1].score ) .map( ([country, item]) => ({ country, name: item.name, flag: item.flag, score: item.score, commentCount: item.commentCount, superChats: item.superChats, latestCommenter: item.latestCommenter }) ); // ================================================= // PLAYER LEADERBOARD // ================================================= const activePlayers = Object.values(players) .filter( player => player.score > 0 ) .sort( (a, b) => b.score - a.score ) .map( player => { const country = player.country; return { key: player.key, displayName: player.displayName, country, countryName: country && info[country] ? info[country][1] : "", flag: country && info[country] ? info[country][0] : "🌍", score: player.score, comments: player.comments, superChats: player.superChats }; } ); // ================================================= // EVENT // ================================================= let event = null; if ( lastEvent && lastEvent.id ) { event = { ...lastEvent, countryName: lastEvent.country && info[lastEvent.country] ? info[lastEvent.country][1] : "", flag: lastEvent.country && info[lastEvent.country] ? info[lastEvent.country][0] : "🌍" }; } // ================================================= // RESPONSE // ================================================= res.json({ success: true, countries: activeCountries, players: activePlayers, event, youtube: { connected: Boolean( liveChatId ), videoId: VIDEO_ID }, settings: { commentPoints: COMMENT_POINTS, superChatPointsPerUSD: SUPERCHAT_POINTS_PER_USD }, serverTime: Date.now() }); } ); // ===================================================== // HEALTH CHECK // ===================================================== app.get( "/api/health", (req, res) => { res.json({ ok: true, youtubeConnected: Boolean( liveChatId ), videoId: VIDEO_ID, time: Date.now() }); } ); // ===================================================== // HOME PAGE // ===================================================== app.get( "/", (req, res) => { res.sendFile( path.join( __dirname, "public", "index.html" ) ); } ); // ===================================================== // 404 // ===================================================== app.use( (req, res) => { res.status(404).json({ success: false, error: "Not Found" }); } ); // ===================================================== // START SERVER // ===================================================== app.listen( PORT, () => { console.log( "======================================" ); console.log( "🌍 COUNTRY BATTLE LIVE" ); console.log( "======================================" ); console.log( `Server running on port ${PORT}` ); console.log( `Video ID: ${VIDEO_ID}` ); console.log( `Comment Points: ${COMMENT_POINTS}` ); console.log( `Super Chat: $1 = ${SUPERCHAT_POINTS_PER_USD} points` ); console.log( "======================================" ); // Start YouTube polling pollChat(); } );
